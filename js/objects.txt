function getObjectData(id){

    const data=objectData[id];

    if(!data){
        return {
            type:"none",
            width:1,
            height:1,
            layer:0,
            tags:[]
        };
    }

    return {
        ...data,
        tags:[
            ...(data.tags||[])
        ]
    };

}

class GameObject{

    constructor(id,x,y,data={}){

        const object=getObjectData(id);

        this.id=id;

        this.x=x;
        this.y=y;

        this.type=
        object.type;

        this.width=
        object.width*tileSize;

        this.height=
        object.height*tileSize;

        this.layer=
        object.layer;

        this.tags=[
            ...object.tags
        ];

        this.active=true;

        this.visible=true;

        this.effects={};

        this.data={
            ...object,
            ...data
        };

    }


    update(){}


    onCollision(){}


    destroy(){

        this.active=false;

    }

}

function createObject(id,x,y,data={}){

    return new GameObject(
        id,
        x,
        y,
        data
    );

}
